{{- $cartoonData := hugo.Data.editorial_cartoons | default dict -}} {{- $cartoons := partial "editorial/published-cartoons.html" ($cartoonData.cartoons | default (slice)) -}} {{- /* Effective release time selects the newest published art; equal instants use slug order. */ -}} {{- $current := false -}} {{- $currentRelease := 0 -}} {{- range $cartoons -}} {{- $release := (time.AsTime (.publishDate | default .date)).UnixNano -}} {{- $newest := not $current -}} {{- if $current -}} {{- $newest = or (gt $release $currentRelease) (and (eq $release $currentRelease) (lt .slug $current.slug)) -}} {{- end -}} {{- if $newest -}} {{- $current = . -}} {{- $currentRelease = $release -}} {{- end -}} {{- end -}} {{- return $current -}}
