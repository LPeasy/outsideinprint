(index $product "title") -}} {{- if ne (index $product "operator_line") $expectedOperatorLine -}} {{- errorf "Apps product products.%s operator_line does not match the approved identity" $key -}} {{- end -}} {{- $supportEmail := "support@outsideinprint.org" -}} {{- if ne (index $product "support_email") $supportEmail -}} {{- errorf "Apps product products.%s support_email must equal %q" $key $supportEmail -}} {{- end -}} {{- $expectedSupportLine := printf "Support for %s is provided by Outside In Print LLC: %s." (index $product "title") $supportEmail -}} {{- if ne (index $product "support_line") $expectedSupportLine -}} {{- errorf "Apps product products.%s support_line does not match the approved LLC support identity" $key -}} {{- end -}} {{- with $page -}} {{- if ne .Title (index $product "title") -}} {{- errorf "Apps page %s title must match products.%s.title" .Path $key -}} {{- end -}} {{- if ne .Slug (index $product "slug") -}} {{- errorf "Apps page %s slug must match products.%s.slug" .Path $key -}} {{- end -}} {{- end -}} {{- return $product -}}
